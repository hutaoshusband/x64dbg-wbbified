settingset Events, TlsCallbacks, 0
settingset Events, EntryBreakpoint, 1
varnew $outer, 0
varnew $inner, 0
varnew $plugin_direct_mode, 1
init tests/scriptcmd_call.exe
mov startup_stops, 0
await_entry:
cmp cip, scriptcmd_call:entry
je entry_ready
inc startup_stops
testassert startup_stops <= .64, "execution fixture entry stop limit"
run
jmp await_entry
entry_ready:
testassert cip == scriptcmd_call:entry, "execution fixture must stop at its actual entry point"
bp scriptcmd_call:OuterHit
SetBreakpointSilent scriptcmd_call:OuterHit, 1
SetBreakpointCondition scriptcmd_call:OuterHit, 0
bp scriptcmd_call:InnerHit
SetBreakpointSilent scriptcmd_call:InnerHit, 1
SetBreakpointCondition scriptcmd_call:InnerHit, 0
run
testassert $outer == 1, "plugin-direct fast-resume outer callback ran exactly once"
testassert $inner == 1, "plugin-direct fast-resume inner callback ran exactly once"
plugindirectassert
ret

onouter:
testassert $outer == 0, "plugin-direct fast-resume outer callback entered once"
mov $outer, 1
ret

oninner:
testassert $inner == 0, "plugin-direct fast-resume inner callback entered once"
mov $inner, 1
ret
